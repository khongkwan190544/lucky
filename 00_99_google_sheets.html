ตั้งค่า Google Sheet Web App URL

กรอก Web App URL ที่ได้จากการ Deploy Google Apps Script ใน Google Sheet เพื่อเปิดใช้งานการเชื่อมต่อและ Sync ข้อมูลแบบ Real-time

หากยังไม่ได้ตั้งค่า ระบบจะบันทึกข้อมูลไว้ใน LocalStorage ของเบราว์เซอร์ชั่วคราว สามารถใช้งานและทดสอบระบบได้ทันที

ขั้นตอนการเชื่อมต่อ Google Sheets (Apps Script)

  1. เปิด Google Sheets ใหม่ขึ้นมา
  2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  3. ลบโค้ดเดิมทั้งหมดในไฟล์ Code.gs แล้ววางโค้ดด้านล่างนี้ลงไป:
function doGet() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data = sheet.getDataRange().getValues();
  let result = { board: {}, prize1: "", prize2: "" };
  
  if (data.length > 1) {
    result.prize1 = String(data[0][1] || "");
    result.prize2 = String(data[0][3] || "");
    
    for (let i = 1; i < data.length; i++) {
      let num = String(data[i][0]).padStart(2, '0');
      let name = data[i][1];
      if (num && name) {
        result.board[num] = name;
      }
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const payload = JSON.parse(e.postData.contents);
  
  if (payload.action === 'save') {
    sheet.clear();
    sheet.appendRow(["Prize1", payload.prize1 || "", "Prize2", payload.prize2 || ""]);
    sheet.appendRow(["Number", "Name"]);
    
    for (let num in payload.board) {
      sheet.appendRow([num, payload.board[num]]);
    }
  } else if (payload.action === 'reset') {
    sheet.clear();
    sheet.appendRow(["Prize1", "", "Prize2", ""]);
    sheet.appendRow(["Number", "Name"]);
  }
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
